// with extra space
class Solution {
    public void rotate(int[][] matrix) {
        int n = matrix.length;
        int[][] result = new int[n][n];

        for(int i = 0;i < n;i++){
            for(int j = n-1;j >= 0;j--){
                result[i][n-1-j] = matrix[j][i];
            }
        }

        for(int i = 0;i < n;i++){
            for(int j = 0;j < n;j++){
                matrix[i][j] = result[i][j];
            }
        }
    }
}

Without extra space
class Solution {
    public void rotate(int[][] matrix) {
        int n = matrix.length;
        
        //Transpose of the matrix
        for(int i = 0;i < n;i++){
            for(int j = i;j < n;j++){
                int temp = matrix[i][j];
                matrix[i][j] = matrix[j][i];
                matrix[j][i] = temp;
            }
        }
        
        //reverse the numbers of a row in the matrix
        for(int i = 0;i < n;i++){
            int p1 = 0;
            int p2 = n-1;

            while(p1 < p2){
                int temp = matrix[i][p1];
                matrix[i][p1] =  matrix[i][p2];
                matrix[i][p2] = temp;
                p1++;
                p2--; 
            }
        }
    }
}
