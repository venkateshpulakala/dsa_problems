//Brute Force Approach
class Solution {
    public int maxRotateFunction(int[] nums) {
        int maxSum = Integer.MIN_VALUE;
        int currSum = 0;
        int k = 0;
        int n = nums.length;
        for(int i = 0;i < n;i++){
            for(int j = 0;j < n;j++){
                int p = (k + j + n)%n;
                currSum += nums[j]*p;
            }
            maxSum = Math.max(maxSum,currSum);
            currSum = 0;
            k++;
        }
        return maxSum;
    }
}

//Time Complexity = O(n^2)
//Space Complexity = O(1)

//Optimal Approach
class Solution {
    public int maxRotateFunction(int[] nums) {
        
        int result = Integer.MIN_VALUE;
        int n = nums.length;
        int f = 0;
        int sum = 0;
        for(int i = 0;i < n;i++){
            sum += nums[i];
            f += i*nums[i];
        }
        result = f;
        for(int i = 0;i <= n-1;i++){
            int newF = f + sum - n*nums[n-i-1];
            result = Math.max(result,newF);
            f = newF;
        }
        return result;
    }
}

//Time Complexity : O(n)
//Space Complexity : O(1)
